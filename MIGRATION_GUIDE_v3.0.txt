3. UNIFIED STREAMING CONNECTOR API
--------------------------------------------------------------------------------
All live connectors implement the unified `LiveConnector` interface:

```typescript
import { LiveConnector, LivePost, ConnectorState } from './src/live';

interface LiveConnector {
  id: string;
  platform: 'reddit' | 'bluesky' | 'rss' | 'x' | 'youtube' | 'instagram' | 'local';
  connect(): Promise<void> | void;
  disconnect(): void;
  onMessage(handler: (post: LivePost) => void): void;
  getStatus(): ConnectorState;
}
```

Usage Example:
```typescript
import { LiveManager } from './src/live';

const liveManager = new LiveManager({
  x: { query: 'AI OR cyber', offline: false, accessToken: 'BEARER_TOKEN' },
  bluesky: { keywords: ['disinformation', 'market'], offline: false },
  reddit: { subreddits: ['technology', 'worldnews'], offline: false },
  onPost: (post) => {
    console.log(`[${post.platform.toUpperCase()}] ${post.authorName}: ${post.content}`);
  }
});

await liveManager.startAll();
```
